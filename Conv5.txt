foreach ($header in $headers) {
    $content = Get-Content $header -Raw
    
    # Add opening guard after #define
    if ($content -notmatch '#ifdef __cplusplus') {
        # Find the last #include
        $lines = $content -split "`n"
        $lastInclude = -1
        for ($i = 0; $i -lt $lines.Count; $i++) {
            if ($lines[$i] -match '^#include') {
                $lastInclude = $i
            }
        }
        
        if ($lastInclude -ge 0) {
            # Insert after last include
            $lines = @($lines[0..$lastInclude]) + @("", "#ifdef __cplusplus", "extern `"C`" {", "#endif", "") + @($lines[($lastInclude+1)..($lines.Count-1)])
            $content = $lines -join "`n"
        }
        
        # Add closing guard before final #endif
        $content = $content -replace '(#endif\s*/[*/]\s*\w+_H\s*)$', "#ifdef __cplusplus`n}`n#endif`n`n`$1"
        
        Set-Content $header -Value $content -NoNewline
        Write-Host "Added extern C guards to $(Split-Path $header -Leaf)"
    }
}
